# ============================================
# CLI урок 2: «SQL в терминале» — пайпы и фильтры
# ============================================

# CSV: week1/cli-practice/data/students.csv
# 6 строк данных + заголовок = 7 строк в файле
# У Егора grade пустой — файловый аналог NULL

# SQL: SELECT city FROM students
$ tail -n +2 week1/cli-practice/data/students.csv | cut -d',' -f3
# Москва, Казань, Москва, Сочи, Казань, Москва

# SQL: WHERE city = 'Москва'
$ grep 'Москва' week1/cli-practice/data/students.csv
# 3 строки (Аня, Вера, Егор)

# SQL: GROUP BY city + COUNT(*)
$ tail -n +2 week1/cli-practice/data/students.csv | cut -d',' -f3 | sort | uniq -c
# 2 Казань, 3 Москва, 1 Сочи

# SQL: ORDER BY grade DESC LIMIT 3
$ sort -t',' -k5 -rn week1/cli-practice/data/students.csv | head -3
# (вставь свой вывод)

# SQL: COUNT(*)
$ wc -l week1/cli-practice/data/students.csv
# 7 (включая заголовок!)

# ============================================
# ТРИ ЛОВУШКИ
# ============================================

# Ловушка 1: заголовок
# wc -l считает заголовок (7 вместо 6). SQL COUNT(*) = 6.
# Чинится: tail -n +2 file | wc -l

# Ловушка 2: Егор и sort
# sort -rn: пустое поле = 0 → Егор ПЕРВЫМ (reverse)
# SQL: порядок NULL зависит от СУБД.
#   DuckDB: DESC → NULLS LAST (Егор в конце)
#   PostgreSQL: DESC → NULLS FIRST (Егор в начале)
# Правильно на собесе: "задаю явно NULLS FIRST/LAST"

# Ловушка 3: sort перед uniq -c
# uniq склеивает ТОЛЬКО соседние строки.
# Без sort: Москва (1), Казань (1), Москва (1), Сочи (1), Казань (1), Москва (1)
# С sort: 2 Казань, 3 Москва, 1 Сочи
